# Git Rail Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a read-only right-hand rail listing commits since the branch left trunk, each
tagged with a review-state badge rolled up from the existing `ReviewLedger`, plus an
"uncommitted work" entry for whatever is still dirty in the working tree.

**Architecture:** A new `CommitLog` port (thin git-log plumbing) and a new `GitRail` port
(the orchestrator) sit alongside the existing `Baseline`/`SnapshotStore`/`ReviewLedger` ports.
`app/gitrail.ts` reuses `app/board.ts`'s existing `chunksOf` unchanged — a commit's own chunks
are computed exactly the way the board computes the whole branch's, just scoped to one
commit's `parent..self` range — then rolls each commit up to one of three states by reading
`ChunkReview.outcome` straight out of the ledger. No new bookkeeping, no new writes, no new
classification logic. The UI pulls it over `GET /git-rail` on the same `diffRevision` signal
`/diff` already uses.

**Tech Stack:** TypeScript, Bun (`bun:test`, `Bun.spawn`), React (no build step beyond `bun
build`), plain CSS with custom properties (no CSS framework).

**Spec:** `docs/superpowers/specs/2026-09-08-git-rail-design.md`

## Global Constraints

- Strictly read-only: no revert/edit endpoint, no new ledger writes (spec Non-goals).
- The review-state rollup reads `ChunkReview.outcome` only — never re-runs the risk bar, never
  re-derives `LiveChunk`/`BoardState` (spec Architecture, "Rollup rule").
- `/git-rail` is a pull route (`GET`, fetched on the `diffRevision` signal), never pushed over
  the WebSocket broadcast (spec "Data flow").
- Layering: `app/gitrail.ts` may only import from `core/` (ports and types) plus `app/board.ts`
  — never a concrete adapter. `tests/architecture.test.ts` enforces this; run it after Task 3.
- Merge commits are handled via first-parent only (`%P`'s first token) — no per-parent
  reconstruction (spec Non-goals).

```ts
import type { CommitEntry, CommitLog } from '../../core/ports.ts'
import { gitOrThrow } from './commands.ts'

/**
 * Git's own commit graph, independent of the ledger or the board.
 *
 * One field per commit, delimited the same way `delta.ts` already delimits
 * `--name-status -z` output — a control character no commit subject is going to contain,
 * rather than something a subject legitimately could (a comma, a pipe, a colon).
 */

const FIELD_SEP = '\x1f'
const FORMAT = `%H${FIELD_SEP}%P${FIELD_SEP}%s`

function parseCommits(raw: string): CommitEntry[] {
  const entries: CommitEntry[] = []
  for (const line of raw.split('\n')) {
    if (line === '') continue
    const [sha, parents, subject] = line.split(FIELD_SEP)
    if (sha === undefined || parents === undefined || subject === undefined) continue
    const parentSha = parents === '' ? null : (parents.split(' ')[0] ?? null)
    entries.push({ sha, parentSha, subject })
  }
  return entries
}

export function createGitCommitLog(options: { cwd: string }): CommitLog {
  const { cwd } = options
  return {
    since: async (base) => {
      // `--format=` behaves like `tformat:` — one newline-terminated record per commit,
      // which is what makes splitting on `\n` safe: a commit subject is one line by
      // definition (`%s` is exactly the subject, never the body).
      const raw = await gitOrThrow(cwd, ['log', `--format=${FORMAT}`, `${base}..HEAD`])
      return parseCommits(raw)
    },
  }
}
```

```ts
import { chunksOf } from './board.ts'
import type { Baseline, CommitLog, GitRail, ReviewLedger, SnapshotStore } from '../core/ports.ts'
import type { Chunk, ChunkReview, FileDelta, GitRailView, RailEntry, RailState } from '../core/types.ts'

/**
 * The right-hand rail's own board: commit history since the branch left trunk, each commit's
 * review state rolled up from the same `ReviewLedger` the board itself reads — no new
 * bookkeeping, no re-running the risk bar. A commit's chunks are computed exactly the way the
 * board computes its own (`chunksOf`, unchanged), just scoped to one commit's own
 * parent-to-self range instead of the whole branch's merge-base-to-working-tree range.
 */

export type GitRailDeps = {
  snapshots: SnapshotStore
  baseline: Baseline
  commitLog: CommitLog
  ledger: ReviewLedger
}

function statsOf(deltas: FileDelta[]): { filesChanged: number; added: number; removed: number } {
  return {
    filesChanged: deltas.length,
    added: deltas.reduce((sum, delta) => sum + delta.addedLines.length, 0),
    removed: deltas.reduce((sum, delta) => sum + delta.removedLines.length, 0),
  }
}

/**
 * One state per commit, read straight off `ChunkReview.outcome` — no new classification.
 * `sent-back` wins outright; `approved` requires every one of the commit's own chunks to
 * carry a decision; anything else (auto-passed, archived, or simply never decided) is
 * `quiet`, matching the design's own three-badge treatment.
 */
export function rollUp(chunks: Chunk[], decided: ReadonlyMap<string, ChunkReview>): RailState {
  const outcomes = chunks.map((chunk) => decided.get(chunk.key)?.outcome ?? null)
  if (outcomes.some((outcome) => outcome === 'sent-back')) return 'sent-back'
  if (
    chunks.length > 0 &&
    outcomes.every((outcome) => outcome === 'approved' || outcome === 'yours')
  ) {
    return 'approved'
  }
  return 'quiet'
}

async function railEntry(
  snapshots: SnapshotStore,
  base: string,
  next: string,
  sha: string | null,
  subject: string,
  deltas: FileDelta[],
  decided: ReadonlyMap<string, ChunkReview>,
): Promise<RailEntry> {
  const chunks = await chunksOf(snapshots, base, next, deltas)
  return { sha, subject, ...statsOf(deltas), state: rollUp(chunks, decided) }
}

export function createGitRail(deps: GitRailDeps): GitRail {
  return {
    view: async (): Promise<GitRailView> => {
      const resolution = await deps.baseline.resolve(null)
      const decided = await deps.ledger.all(resolution.branch)

      const commits: RailEntry[] = []
      if (resolution.mergeBaseCommit !== null) {
        const log = await deps.commitLog.since(resolution.mergeBaseCommit)
        for (const commit of log) {
          // Every commit `since()` returns has `mergeBaseCommit` as an ancestor, so it
          // always has a parent — a null one would mean a `CommitLog` bug, not a real repo
          // state. Skipped rather than guessed at: `app/` has no portable way to name an
          // empty tree (that's an `adapters/git` concern).
          if (commit.parentSha === null) continue
          const deltas = await deps.snapshots.delta(commit.parentSha, commit.sha)
          commits.push(
            await railEntry(
              deps.snapshots,
              commit.parentSha,
              commit.sha,
              commit.sha,
              commit.subject,
              deltas,
              decided,
            ),
          )
        }
      }

      const head = await deps.snapshots.head()
      const next = await deps.snapshots.capture()
      const workingDeltas = await deps.snapshots.delta(head, next)
      const workingTree =
        workingDeltas.length === 0
          ? null
          : await railEntry(
              deps.snapshots,
              head,
              next,
              null,
              'Uncommitted work in the tree',
              workingDeltas,
              decided,
            )

      return {
        branch: resolution.branch,
        trunk: resolution.trunk,
        atBranchPoint: resolution.mergeBaseCommit !== null,
        commits,
        workingTree,
      }
    },
  }
}
```
